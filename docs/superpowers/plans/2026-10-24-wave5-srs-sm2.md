  export type Grade = "again" | "hard" | "good" | "easy";
  export type SrsStatus = "new" | "learning" | "review" | "relearning";
  export type SrsItem = { key: string; status: SrsStatus; dueAt: number | null;
    reviewCount: number; lastReviewedAt: number | null; updatedAt: number;
    ease: number; intervalDays: number; lapses: number };
  export const SRS_MATURE_DAYS = 21;
  export const MIN_EASE = 1.3, MAX_EASE = 3.0, INIT_EASE = 2.5;

  function clampEase(e: number): number { return Math.max(MIN_EASE, Math.min(MAX_EASE, e)); }

  export function applyGrade(it: SrsItem, grade: Grade, now: number): SrsItem {
    const reviewCount = it.reviewCount + 1;
    let { status, ease, intervalDays, lapses } = it;
    const learning = status === "new" || status === "learning";
    if (learning) {
      if (grade === "again")   return fin("learning", ease, 0, now + 60_000);
      if (grade === "hard")    return fin("learning", ease, 0, now + 10 * 60_000);
      if (grade === "good")    return fin("review", ease, 1, now + DAY);
      return fin("review", clampEase(ease + 0.15), 4, now + 4 * DAY);
    }
    if (status === "relearning") {
      if (grade === "again") return fin("relearning", clampEase(ease - 0.2), intervalDays, now + 60_000);
      if (grade === "hard")  return fin("relearning", clampEase(ease - 0.15), intervalDays, now + 10 * 60_000);
      if (grade === "good")  return fin("review", ease, Math.max(1, intervalDays), now + Math.max(1, intervalDays) * DAY);
      return fin("review", clampEase(ease + 0.15), Math.max(1, Math.floor(intervalDays * 1.5)), now + Math.max(1, Math.floor(intervalDays * 1.5)) * DAY);
    }
    // review
    if (grade === "again") {
      const iv = Math.max(1, Math.floor(0.5 * intervalDays));
      return fin("relearning", clampEase(ease - 0.2), iv, now + 10 * 60_000, lapses + 1);
    }
    const easeNext = grade === "hard" ? clampEase(ease - 0.15)
      : grade === "easy" ? clampEase(ease + 0.15) : ease;
    const iv = grade === "hard" ? Math.max(1, Math.floor(1.2 * intervalDays))
      : grade === "good" ? Math.max(1, Math.floor(intervalDays * ease))
      : Math.max(1, Math.floor(intervalDays * ease * 1.3));
    return fin(status, easeNext, iv, now + iv * DAY, lapses);
  }
